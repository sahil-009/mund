#include <vector>
#include <string>
// FORGE Editor — C++ Demo
int factorial(int n) {
if (n <= 1) return 1;
return factorial(n - 1) * n;
}
int main() {
std::cout << "Hello from FORGE!\n";
std::vector<int> nums = {1, 2, 3, 4, 5};
for (const auto& n : nums) {
std::cout << n << "! = " << factorial(n);
}
return 0;
}
